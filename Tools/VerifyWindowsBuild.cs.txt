using System;using System.IO;using System.Linq;using UnityEditor;using UnityEditor.Build.Reporting;
public static class VerifyWindowsBuild { public static void Run(){Directory.CreateDirectory("Builds/ReviewWindows");var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=EditorBuildSettings.scenes.Where(s=>s.enabled).Select(s=>s.path).ToArray(),locationPathName="Builds/ReviewWindows/BounceClub.exe",target=BuildTarget.StandaloneWindows64,options=BuildOptions.Development});File.WriteAllText("Tools/windows-build-review.txt",report.summary.result+"; errors="+report.summary.totalErrors+"; warnings="+report.summary.totalWarnings+"; bytes="+report.summary.totalSize+"; duration="+report.summary.totalTime);if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Windows build failed");}}
